cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(HidAccess VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HIDACCESS_ASAN "User-mode AddressSanitizer validation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA "/PDBALTPATH:%_PDB%")
if(HIDACCESS_ASAN)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(HidAccess SDK/HidAccess.cpp)
target_include_directories(HidAccess PUBLIC SDK)
target_link_libraries(HidAccess PUBLIC advapi32)

# Public clients never build, download or install a kernel driver implicitly.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/App/Main.cpp")
    add_executable(HidAccessApp WIN32 App/Main.cpp)
    target_link_libraries(HidAccessApp PRIVATE HidAccess user32 gdi32 comctl32 comdlg32 shell32 ole32 uxtheme shlwapi windowscodecs)
    target_link_options(HidAccessApp PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Examples/access.cpp")
    add_executable(HidAccessExample Examples/access.cpp)
    target_link_libraries(HidAccessExample PRIVATE HidAccess)
    target_link_options(HidAccessExample PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Tests/Protocol.cpp")
    add_executable(HidAccessTests Tests/Protocol.cpp)
    target_link_libraries(HidAccessTests PRIVATE advapi32)
    enable_testing()
    add_test(NAME OfflineProtocolValidation COMMAND HidAccessTests)
endif()

set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed HID Access Control package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
    enable_language(RC)
    set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
    set(_resources "")
    set(_id 101)
    foreach(_name TitanHidAccess.inf TitanHidAccess.cat TitanHidAccess.sys TitanHidAccessClock.sys)
        if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${_name}")
            message(FATAL_ERROR "Missing opaque package file: ${_name}")
        endif()
        string(APPEND _resources "${_id} RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/${_name}\"\n")
        math(EXPR _id "${_id}+1")
    endforeach()
    file(WRITE "${PACKAGE_RESOURCE}" "${_resources}")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Installer/Setup.cpp")
    add_library(HidAccessSetupCore Installer/Setup.cpp)
    target_link_libraries(HidAccessSetupCore PUBLIC HidAccess setupapi newdev wintrust crypt32 advapi32 ole32 user32 shell32)
    add_executable(HidAccessSetupTests Tests/SetupOffline.cpp)
    target_link_libraries(HidAccessSetupTests PRIVATE HidAccessSetupCore)
    add_test(NAME OfflineInstallerValidation COMMAND HidAccessSetupTests)
    add_executable(HidAccessSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
    add_executable(HidAccessSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
    foreach(_target HidAccessSetup HidAccessSetupConsole)
        target_link_libraries(${_target} PRIVATE HidAccessSetupCore gdi32 comctl32 comdlg32)
        target_link_options(${_target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    endforeach()
endif()
